BEGIN

// Variable initialisation
SET pizzaChoices AS DICTIONARY 
SET toppingChoices AS DICTIONARY
SET sideChoices AS DICTIONARY
SET cart AS LIST

CLASS Item
	CONSTRUCTOR
		SET self.name TO name
		SET self.price TO price 
	END CONSTRUCTOR
	
FUNCTION getDetails()
	RETURN name + “£” + FORMAT(price. 2dp)
END FUNCTION
END CLASS

CLASS Pizza INHERITS Item
	CONSTRUCTOR(pizzaType, pizzaSize, toppingSelection)
		SET basePrice TO pizzaChoices[pizzaType]
SET pizzaPrice TO (basePrice * pizzaSize)  + SUM(toppingSelection FOR EACH topping in toppingList)
		CALL SUPER CONSTRUCTOR WITH (pizzaType, pizzaPrice)
		SET self.size TO size
SET self.selectedToppings TO toppingList
END CONSTRUCTOR

FUNCTION getDetails()
RETURN FORMATED STRING WITH pizzaType, pizzaSize, toppingSelection, pizzaPrice
	END FUNCTION
END CLASS

CLASS sideOrder INHERITS Item
	CONSTRUCTOR(name, quantity)
		SET totalSidePrice TO sideChoices[name] * quantity
		CALL SUPER CONSTRUCTOR WITH (name, totalSidePrice)
SET self.quantity TO quantity
	END CONSTRUCTOR

	FUNCTION getDetails()
		RETURN FORMATED STRING WITH sideName, quantity, sidePrice
	END FUNCTION
END CLASS

CLASS sideOrderGUI
	CONSTRUCTOR(root)
		SET self.window TO root
		SET windowGeometry TO 500 by 400 pixels
		SET windowTitle TO “Side Orders”
		SET self.selectedSides AS DICTIONARY
	END CONSTRUCTOR
	
	FUNCTION validate()
		FOR EACH side IN sideChoices
		GET quantity FROM SPINBOX
		IF quantity > 0 THEN
			CREATE new SideOrder object 
			APPEND SideOrder object TO cart
			DESTROY window
		END IF
	END FUNCTION

CLASS OrderSummaryGUI

		SET windowGeometry TO 500 by 400 pixels
		SET windowTitle TO “Order Summary”
	END CONSTRUCTOR

CLASS PizzaGUI

		SET window TO root
		SET windowGeometry TO 500 by 400 pixels
		SET windowTitle TO “Your way Pizza Parlour”
		DECLARE self.pizzaTypeVar AS STRING
		DECLARE self.pizzaSizeVar AS INTEGER
		SET toppingSelection AS DICTIONARY
	END CONSTRUCTOR
	
	FUNCTION addToCart()
		TRY
		IF window HAS ATTRIBUTE errorLabel THEN
			DESTROY errorLabel
		IF pizzaType == “--Select Pizza Type–” THEN
			DISPLAY “Please Select pizza type” 
		ELSE
		FOR topping IN toppingChoices
			IF topping IS CHECKED THEN 
				self.toppingSelection[topping] = TRUE
				END IF 
END FOR

GET pizzaType FROM COMBOBOX
GET pizzaSize FROM SPINBOX
CREATE new Pizza object

CREATE root AS TK ROOT WINDOW  
CREATE new PizzaGUI object USING root
